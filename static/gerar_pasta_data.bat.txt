@echo off
setlocal EnableExtensions EnableDelayedExpansion
cd /d "%~dp0"
title Restaurar o banco de dados recuperado
rem ESTE ARQUIVO E UM .BAT E UM .ZIP AO MESMO TEMPO (o .zip com os .sql vem colado depois do ultimo "exit /b").
rem Dois cliques nele: cria a pasta "banco de dados recuperado" ao lado dele, com
rem     sql\   os .sql do banco (um por banco)
rem     data\  a pasta data pronta para o MySQL (se houver MySQL instalado)
rem Para a pasta data sobe uma copia TEMPORARIA do MySQL, so por pipe do Windows (sem porta de rede), numa pasta nova. O MySQL e os
rem bancos instalados no computador nao sao tocados. Sem MySQL instalado so entrega a pasta sql.

set "SELF=%~f0"
set "HERE=%~dp0"
echo.
echo ============================================================
echo  Restaurar o banco de dados recuperado
echo  Cria a pasta "banco de dados recuperado" com os .sql e,
echo  se houver MySQL instalado, a pasta data pronta.
echo  Os seus bancos instalados nao sao tocados.
echo ============================================================
echo.

rem ---------------------------------------------------------------- 1. a pasta de saida e os .sql
where tar >nul 2>&1
if errorlevel 1 (
  echo Este Windows nao tem o comando tar ^(vem no Windows 10 versao 1803 ou mais novo^).
  echo Este arquivo e um .zip por baixo: abra com o 7-Zip ou o WinRAR e extraia os .sql.
  goto :fim_sem_pasta
)
rem ATENCAO: nada de "goto" para tras (rotulo acima do goto) neste arquivo: o cmd procura o rotulo seguindo ate o fim do arquivo,
rem e depois do ultimo "exit /b" vem o .zip (binario), que quebra a busca. Os lacos usam for /l.
set "ROOT=%HERE%banco de dados recuperado"
for /l %%n in (2,1,300) do if exist "!ROOT!" set "ROOT=%HERE%banco de dados recuperado %%n"
set "SQLDIR=!ROOT!\sql"
set "OUT=!ROOT!\data"
mkdir "!SQLDIR!" || (echo Nao consegui criar a pasta !ROOT!. & goto :fim_sem_pasta)
set "LOG=!ROOT!\gerar_pasta_data.log"
echo Inicio %date% %time% > "!LOG!"
echo Extraindo os .sql para a pasta sql ...
tar -xf "!SELF!" -C "!SQLDIR!" 2>> "!LOG!"
if errorlevel 1 (
  echo Nao consegui extrair os arquivos deste pacote. Veja !LOG!.
  goto :fim
)
set "NSQL=0"
for %%f in ("!SQLDIR!\*.sql") do set /a NSQL+=1
if "!NSQL!"=="0" (
  echo Nao ha nenhum .sql no pacote.
  goto :fim
)
echo Pronto: !NSQL! arquivo^(s^) .sql em !SQLDIR!

rem ---------------------------------------------------------------- 2. o MySQL instalado
set "MYSQLD="
set "INI="
for %%s in (MySQL MySQL55 MySQL56 MySQL57 MySQL51 MariaDB) do if not defined MYSQLD call :servico %%s
if not defined MYSQLD for /d %%d in ("%ProgramFiles%\MySQL\MySQL Server 5.*" "%ProgramW6432%\MySQL\MySQL Server 5.*" "%ProgramFiles(x86)%\MySQL\MySQL Server 5.*" "C:\MySQL" "C:\xampp\mysql" "C:\wamp\bin\mysql\*") do if not defined MYSQLD if exist "%%~d\bin\mysqld.exe" set "MYSQLD=%%~d\bin\mysqld.exe"
if not defined MYSQLD for /f "delims=" %%m in ('where mysqld.exe 2^>nul') do if not defined MYSQLD set "MYSQLD=%%m"
if not defined MYSQLD (
  echo.
  echo O MySQL nao esta instalado neste computador, entao nao da para gerar a pasta data.
  echo Os .sql ja estao em !SQLDIR! ^(la tem o restaurar.bat para importar num MySQL^).
  echo MySQL nao encontrado >> "!LOG!"
  goto :fim
)
for %%i in ("!MYSQLD!") do set "BIN=%%~dpi"
for %%i in ("!BIN!..") do set "BASE=%%~fi"
if not exist "!BIN!mysql.exe" (
  echo Falta o mysql.exe ao lado do mysqld.exe em !BIN!. Instale o MySQL completo.
  goto :fim
)
set "VER="
for /f "delims=" %%l in ('""!MYSQLD!" --version"') do set "L=%%l"
set "R=!L:*  Ver =!"
for /f %%v in ("!R!") do set "VER=%%v"
echo MySQL: !MYSQLD!
echo Versao: !VER!
echo MySQL !MYSQLD! !VER! >> "!LOG!"
if not "!VER:~0,3!"=="5.5" echo AVISO: a pasta gerada serve para o MySQL !VER!. Se o sistema usa outra versao, gere com a mesma versao dele.

rem ---------------------------------------------------------------- 3. configuracao do MySQL do computador
if not defined INI if exist "%ProgramData%\MySQL\MySQL Server 5.5\my.ini" set "INI=%ProgramData%\MySQL\MySQL Server 5.5\my.ini"
if not defined INI if exist "!BASE!\my.ini" set "INI=!BASE!\my.ini"
if not defined INI if exist "%windir%\my.ini" set "INI=%windir%\my.ini"
set "CS=latin1"
set "LCTN="
call :chave character-set-server CSV
if not defined CSV call :chave character_set_server CSV
if defined CSV set "CS=!CSV!"
call :chave lower_case_table_names LCTN
set "EXTRA="
if defined LCTN set "EXTRA=--lower_case_table_names=!LCTN!"
if defined INI (echo Configuracao lida de: !INI!) else echo Configuracao do MySQL nao encontrada: usando o padrao.
echo Conjunto de caracteres: !CS!

rem ---------------------------------------------------------------- 4. a pasta data nova
mkdir "!OUT!" || (echo Nao consegui criar a pasta !OUT!. & goto :fim)
set "PIPE=RecuperadorTmp%RANDOM%"

if exist "%HERE%mysql_original\user.frm" (
  echo Usando o esquema mysql ^(usuarios e senhas^) da pasta mysql_original.
  xcopy "%HERE%mysql_original" "!OUT!\mysql\" /e /i /q /y >nul
) else if exist "!BASE!\data\mysql\user.frm" (
  xcopy "!BASE!\data\mysql" "!OUT!\mysql\" /e /i /q /y >nul
) else (
  call :criar_mysql
  if errorlevel 1 (echo Nao consegui criar o esquema mysql. & goto :falhou)
)
if exist "!BASE!\data\performance_schema" xcopy "!BASE!\data\performance_schema" "!OUT!\performance_schema\" /e /i /q /y >nul

rem ---------------------------------------------------------------- 5. sobe o MySQL temporario
echo.
echo Subindo o MySQL temporario ...
start "MySQL temporario - nao feche" /min "!MYSQLD!" --no-defaults --basedir="!BASE!" --datadir="!OUT!" --pid-file="!OUT!\recuperador.pid" --log-error="!OUT!\recuperador.err" --enable-named-pipe --socket=!PIPE! --skip-networking --skip-grant-tables --default-storage-engine=INNODB --character-set-server=!CS! --innodb_fast_shutdown=0 --innodb_flush_log_at_trx_commit=0 --innodb_buffer_pool_size=256M --innodb_log_file_size=48M --max_allowed_packet=256M !EXTRA!
set "PRONTO="
for /l %%t in (1,1,120) do if not defined PRONTO (
  "!BIN!mysqladmin.exe" --no-defaults --protocol=pipe -S !PIPE! ping >nul 2>&1 && set "PRONTO=1"
  if not defined PRONTO ping -n 2 127.0.0.1 >nul
)
if not defined PRONTO (
  echo O MySQL temporario nao subiu em 2 minutos. Veja o final de "!OUT!\recuperador.err".
  goto :falhou
)
echo MySQL temporario no ar.

rem ---------------------------------------------------------------- 6. importa os .sql
set "ERROS=0"
for %%f in ("!SQLDIR!\*.sql") do (
  echo Importando %%~nxf ^(pode demorar alguns minutos^) ...
  echo Importando %%~nxf >> "!LOG!"
  "!BIN!mysql.exe" --no-defaults --protocol=pipe -S !PIPE! --max_allowed_packet=256M --default-character-set=utf8 < "%%f" 2>> "!LOG!"
  if errorlevel 1 (
    set /a ERROS+=1
    echo   ERRO ao importar %%~nxf. Veja gerar_pasta_data.log.
  ) else echo   Ok.
)

rem ---------------------------------------------------------------- 7. confere o que entrou
echo.
echo Conferencia do que entrou no MySQL:
"!BIN!mysql.exe" --no-defaults --protocol=pipe -S !PIPE! -N -e "select concat('  banco ',table_schema,': ',sum(table_type='BASE TABLE'),' tabelas, ',sum(table_type='VIEW'),' views') from information_schema.tables where table_schema not in ('mysql','information_schema','performance_schema') group by table_schema" 2>> "!LOG!"
"!BIN!mysql.exe" --no-defaults --protocol=pipe -S !PIPE! -N -e "select concat('  gatilhos: ',count(*)) from information_schema.triggers" 2>> "!LOG!"
"!BIN!mysql.exe" --no-defaults --protocol=pipe -S !PIPE! -N -e "select concat('  chaves estrangeiras: ',count(*)) from information_schema.table_constraints where constraint_type='FOREIGN KEY'" 2>> "!LOG!"
set "NTAB=0"
for /f %%n in ('findstr /b /c:"CREATE TABLE" "!SQLDIR!\*.sql" ^| find /c /v ""') do set "NTAB=%%n"
echo   CREATE TABLE nos .sql: !NTAB!

rem ---------------------------------------------------------------- 8. desliga limpo e espera soltar os arquivos
echo.
echo Desligando o MySQL temporario ...
"!BIN!mysqladmin.exe" --no-defaults --protocol=pipe -S !PIPE! shutdown >nul 2>&1
set "SOLTOU="
for /l %%t in (1,1,600) do if not defined SOLTOU (
  (>>"!OUT!\ibdata1" type nul) 2>nul && if not exist "!OUT!\recuperador.pid" set "SOLTOU=1"
  if not defined SOLTOU ping -n 2 127.0.0.1 >nul
)
if not defined SOLTOU (
  echo O MySQL temporario nao desligou em 10 minutos. Nao use esta pasta.
  goto :falhou
)
rem Desligado de forma limpa: os logs do InnoDB estao vazios. Apagar deixa o MySQL do sistema recriar no tamanho que o my.ini dele pede.
del /q "!OUT!\ib_logfile*" 2>nul
del /q "!OUT!\recuperador.pid" 2>nul
if exist "!OUT!\recuperador.err" move /y "!OUT!\recuperador.err" "!ROOT!\mysql_temporario.log" >nul
echo.
echo ============================================================
if "!ERROS!"=="0" (echo  PRONTO.) else echo  Terminou com !ERROS! erro^(s^) na importacao. Veja gerar_pasta_data.log.
echo  Pasta: !ROOT!
echo    sql   os .sql do banco
echo    data  a pasta data pronta para o MySQL
echo.
echo  Para usar a data no lugar do banco atual: pare o servico do
echo  MySQL, guarde a pasta data antiga, coloque esta no lugar e
echo  inicie o servico de novo.
echo ============================================================
goto :fim

:falhou
echo.
echo Falhou. A pasta !OUT! esta incompleta: nao use. Os .sql continuam em !SQLDIR!. Detalhes em gerar_pasta_data.log.
"!BIN!mysqladmin.exe" --no-defaults --protocol=pipe -S !PIPE! shutdown >nul 2>&1
echo FALHOU >> "!LOG!"

:fim
echo.
pause
endlocal
exit /b 0

:fim_sem_pasta
echo.
pause
endlocal
exit /b 1

rem ---------------------------------------------------------------- sub-rotinas
:servico
set "IMG="
for /f "tokens=2,*" %%a in ('reg query "HKLM\SYSTEM\CurrentControlSet\Services\%1" /v ImagePath 2^>nul ^| findstr /i "ImagePath"') do set "IMG=%%b"
if not defined IMG goto :eof
set "T=!IMG:"=#!"
set "EXE=" & set "OPT=" & set "CFG="
for /f "tokens=1,2,3 delims=#" %%p in ("!T!") do (set "EXE=%%p" & set "OPT=%%q" & set "CFG=%%r")
if not exist "!EXE!" if exist "!EXE!.exe" set "EXE=!EXE!.exe"
if not exist "!EXE!" goto :eof
set "MYSQLD=!EXE!"
if "!OPT:defaults-file=!"=="!OPT!" goto :eof
if exist "!CFG!" set "INI=!CFG!"
goto :eof

:chave
set "%~2="
if not defined INI goto :eof
if not exist "!INI!" goto :eof
set "V="
for /f "usebackq tokens=1,* delims==" %%a in (`findstr /b /i /c:"%~1" "!INI!"`) do set "V=%%b"
if not defined V goto :eof
set "V=!V:"=!"
for /f "tokens=1" %%x in ("!V!") do set "%~2=%%x"
goto :eof

:criar_mysql
rem Instalacao sem o esquema pronto: cria pelo roteiro do proprio MySQL (share\mysql_system_tables*.sql).
if not exist "!BASE!\share\mysql_system_tables.sql" exit /b 1
mkdir "!OUT!\mysql" 2>nul
set "BOOT=%TEMP%\recuperador_boot_%RANDOM%.sql"
(echo create database if not exists mysql; & echo use mysql; & echo set @current_hostname='localhost'; & type "!BASE!\share\mysql_system_tables.sql" & type "!BASE!\share\mysql_system_tables_data.sql") > "!BOOT!"
"!MYSQLD!" --no-defaults --bootstrap --basedir="!BASE!" --datadir="!OUT!" --default-storage-engine=MyISAM --skip-innodb < "!BOOT!" >> "!LOG!" 2>&1
set "RC=!errorlevel!"
del /q "!BOOT!" 2>nul
exit /b !RC!
